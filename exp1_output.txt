-- 1b-1. Implement the tables using the given constraints

SQL> CREATE TABLE student (
  2    name           VARCHAR2(20),
  3    student_number NUMBER(3),
  4    class          NUMBER(1),
  5    major          VARCHAR2(10) NOT NULL,
  6    CONSTRAINT pk_student PRIMARY KEY (student_number)
  7  );
Table created.

SQL> CREATE TABLE course (
  2    course_name   VARCHAR2(30),
  3    course_number VARCHAR2(15),
  4    credit_hours  NUMBER(2) NOT NULL,
  5    department    VARCHAR2(15),
  6    CONSTRAINT pk_course PRIMARY KEY (course_number)
  7  );
Table created.

SQL> CREATE TABLE section (
  2    section_identifier NUMBER(4),
  3    course_number      VARCHAR2(15),
  4    semester           VARCHAR2(10) NOT NULL,
  5    year               VARCHAR2(4),
  6    instructor         VARCHAR2(15),
  7    CONSTRAINT pk_section PRIMARY KEY (section_identifier)
  8  );
Table created.

SQL> CREATE TABLE grade_report (
  2    student_number     NUMBER(3),
  3    section_identifier NUMBER(4),
  4    grade              VARCHAR2(5) NOT NULL,
  5    CONSTRAINT pk_grade_report PRIMARY KEY (student_number, section_identifier),
  6    CONSTRAINT fk_gr_student FOREIGN KEY (student_number)
  7      REFERENCES student(student_number) ON DELETE CASCADE,
  8    CONSTRAINT fk_gr_section FOREIGN KEY (section_identifier)
  9      REFERENCES section(section_identifier) ON DELETE CASCADE
 10  );
Table created.

SQL> CREATE TABLE prerequisite (
  2    course_number       VARCHAR2(15),
  3    prerequisite_number VARCHAR2(20),
  4    CONSTRAINT pk_prerequisite PRIMARY KEY (course_number, prerequisite_number),
  5    CONSTRAINT fk_prereq_course FOREIGN KEY (course_number)
  6      REFERENCES course(course_number) ON DELETE CASCADE
  7  );
Table created.
